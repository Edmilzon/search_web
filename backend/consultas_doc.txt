                (STR(?marca) AS ?marcaAlimento)
WHERE {
  ?m a :Mascota .
  ?m :nombreMascota ?nomM .

  OPTIONAL { ?m :tieneRaza ?r . }
  OPTIONAL { ?r :perteneceAEspecie ?e . }
  OPTIONAL { ?e :nombreEspecie ?esp . }

  OPTIONAL { ?m :consume ?a . }
  OPTIONAL { ?a :marcaAlimento ?marca . }

  FILTER(?esp = "Perro")
}

PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

SELECT DISTINCT ?nombreMascota ?cuidado
WHERE {
  ?m rdf:type :Mascota ;
     :nombreMascota ?nom .
  ?m :perteneceAEspecie ?especie . 
  ?especie :nombreEspecie ?nE .
  FILTER(STR(?nE) = "Gato") 
  ?m :requiereCuidado ?c .
  ?c :tipoCuidado ?tipo .
  BIND(STR(?nom) AS ?nombreMascota)
  BIND(STR(?tipo) AS ?cuidado)
}

WHERE {
  ?m rdf:type :Mascota .
  
  # Obtenemos el nombre y le quitamos el formato técnico
  ?m :nombreMascota ?nombre .
  BIND(STR(?nombre) AS ?nombreLimpio)
  
  # Obtenemos el cuidado y le quitamos el formato técnico
  ?m :requiereCuidado ?c .
  ?c :tipoCuidado ?tipo .
  BIND(STR(?tipo) AS ?cuidadoLimpio)
}
